    class Segment
    {
        public string Kind = "data";   // head/id/seq/len/data/resv/sum/tail
        public string Tag = "";        // resv 的显示名：帧序号/帧ID/3Byte/定长数组/变长数组
        public int Bytes = 1;
        public string Hex = "";        // head / tail
        public string Name = "";
        public string PType = "uint8"; // data 的插件类型
        public string Endian = "little";
        public double Scale = 1.0;
        public double Bias = 0.0;
        public bool Export = true;     // 是否输出为通道（显示到面板）
        public int LenSize = 2;        // 帧长度字段宽度
        public int LenAdjust = 0;
        public string ChkType = "sum8";
        public int RangeFrom = 0;
        public int RangeTo = -1;
        public int ElemCount = 8;      // 定长数组元素个数
    }

    static class PTypes
    {
        public static int Width(string t)
        {
            switch (t)
            {
                case "int8": case "uint8": return 1;
                case "int16": case "uint16": return 2;
                case "int32": case "uint32": case "float32": return 4;
                case "int64": case "uint64": case "float64": return 8;
            }
            return 0;
        }
        public static string[] ForWidth(int w)
        {
            if (w == 1) return new string[] { "uint8", "int8" };
            if (w == 2) return new string[] { "uint16", "int16" };
            if (w == 4) return new string[] { "uint32", "int32", "float32" };
            if (w == 8) return new string[] { "uint64", "int64", "float64" };
            return new string[] { "uint8" };
        }
        public static string Label(string t)
        {
            switch (t)
            {
                case "uint8": return "无符号整数 (uint8)";
                case "int8": return "有符号整数 (int8)";
                case "uint16": return "无符号整数 (uint16)";
                case "int16": return "有符号整数 (int16)";
                case "uint32": return "无符号整数 (uint32)";
                case "int32": return "有符号整数 (int32)";
                case "float32": return "浮点数 (float32)";
                case "uint64": return "无符号整数 (uint64)";
                case "int64": return "有符号整数 (int64)";
                case "float64": return "浮点数 (float64)";
            }
            return t;
        }
        public static string[] Checksums = new string[] { "none", "sum8", "xor8", "crc16_modbus" };
        public static string ChkLabel(string t)
        {
            switch (t)
            {
                case "none": return "NONE（不校验）";
                case "sum8": return "SUM8";
                case "xor8": return "XOR8";
                case "crc16_modbus": return "CRC16-MODBUS";
            }
            return t;
        }
        public static int ChkBytes(string t) { return t == "crc16_modbus" ? 2 : 1; }
    }

    static class HexUtil
    {
        public static bool IsHexChar(char c)
        {
            return (c >= '0' && c <= '9') || (c >= 'a' && c <= 'f') || (c >= 'A' && c <= 'F');
        }

        // "aa55eb90" / "0xAA,0x55" -> "AA 55 EB 90"
        public static string Normalize(string s)
        {
            if (s == null) return "";
            string t = s.Replace("0x", "").Replace("0X", "");
            StringBuilder sb = new StringBuilder();
            for (int i = 0; i < t.Length; i++)
            {
                char c = t[i];
                if (IsHexChar(c)) sb.Append(char.ToUpperInvariant(c));
                else if (c == ' ' || c == '\t' || c == ',' || c == ';' || c == '_' || c == '-' || c == '\r' || c == '\n') continue;
                else throw new FormatException("非法十六进制字符：" + c);
            }
            if (sb.Length % 2 != 0) throw new FormatException("十六进制位数必须为偶数");
            StringBuilder o = new StringBuilder();
            for (int i = 0; i < sb.Length; i += 2)
            {
                if (i > 0) o.Append(' ');
                o.Append(sb[i]); o.Append(sb[i + 1]);
            }
            return o.ToString();
        }

        public static int ByteCount(string normalized)
        {
            if (normalized == null || normalized.Length == 0) return 0;
            return (normalized.Length + 1) / 3;
        }
    }

    // ================= 帧布局 =================
    class Frame
    {
        public string Name = "MyProtocol";
        public List<Segment> Segs = new List<Segment>();

        public int TotalBytes()
        {
            int n = 0;
            foreach (Segment s in Segs) n += s.Bytes;
            return n;
        }

        public int OffsetOf(int index)
        {
            int n = 0;
            for (int i = 0; i < index && i < Segs.Count; i++) n += Segs[i].Bytes;
            return n;
        }

        public bool HasVarArray()
        {
            foreach (Segment s in Segs) if (s.Kind == "resv" && s.Tag == "变长数组") return true;
            return false;
        }

        public string KindName(Segment s)
        {
            switch (s.Kind)
            {
                case "head": return "帧头";
                case "tail": return "帧尾";
                case "seq": return "帧序号";
                case "id": return "帧ID";
                case "len": return "帧长度";
                case "sum": return "校验";
                case "resv": return s.Tag.Length > 0 ? s.Tag : "保留";
                case "data": return s.Name.Length > 0 ? s.Name : "字段";
            }
            return s.Kind;
        }

        public string ZeroText(int bytes)
        {
            StringBuilder sb = new StringBuilder();
            for (int i = 0; i < bytes; i++)
            {
                if (i > 0) sb.Append(' ');
                sb.Append("00");
            }
            return sb.ToString();
        }
    }

    // ================= 极简 JSON =================
    static class MiniJson
    {
        public static object Parse(string s)
        {
            int i = 0;
            object v = ParseValue(s, ref i);
            SkipWs(s, ref i);
            if (i < s.Length) throw new FormatException("JSON 末尾存在多余内容（位置 " + i + "）");
            return v;
        }
        static void SkipWs(string s, ref int i) { while (i < s.Length && char.IsWhiteSpace(s[i])) i++; }
        static object ParseValue(string s, ref int i)
        {
            SkipWs(s, ref i);
            if (i >= s.Length) throw new FormatException("JSON 意外结束");
            char c = s[i];
            if (c == '{') return ParseObject(s, ref i);
            if (c == '[') return ParseArray(s, ref i);
            if (c == '"') return ParseString(s, ref i);
            if (c == 't') { Expect(s, ref i, "true"); return true; }
            if (c == 'f') { Expect(s, ref i, "false"); return false; }
            if (c == 'n') { Expect(s, ref i, "null"); return null; }
            return ParseNumber(s, ref i);
        }
        static void Expect(string s, ref int i, string lit)
        {
            if (i + lit.Length > s.Length || s.Substring(i, lit.Length) != lit) throw new FormatException("非法字面量（位置 " + i + "）");
            i += lit.Length;
        }
        static Dictionary<string, object> ParseObject(string s, ref int i)
        {
            Dictionary<string, object> d = new Dictionary<string, object>();
            i++; SkipWs(s, ref i);
            if (i < s.Length && s[i] == '}') { i++; return d; }
            while (true)
            {
                SkipWs(s, ref i);
                if (i >= s.Length || s[i] != '"') throw new FormatException("对象的键必须是字符串（位置 " + i + "）");
                string k = ParseString(s, ref i);
                SkipWs(s, ref i);
                if (i >= s.Length || s[i] != ':') throw new FormatException("对象缺少冒号（位置 " + i + "）");
                i++;
                d[k] = ParseValue(s, ref i);
                SkipWs(s, ref i);
                if (i < s.Length && s[i] == ',') { i++; continue; }
                if (i < s.Length && s[i] == '}') { i++; return d; }
                throw new FormatException("对象缺少 , 或 }（位置 " + i + "）");
            }
        }
        static object[] ParseArray(string s, ref int i)
        {
            List<object> a = new List<object>();
            i++; SkipWs(s, ref i);
            if (i < s.Length && s[i] == ']') { i++; return a.ToArray(); }
            while (true)
            {
                a.Add(ParseValue(s, ref i));
                SkipWs(s, ref i);
                if (i < s.Length && s[i] == ',') { i++; continue; }
                if (i < s.Length && s[i] == ']') { i++; return a.ToArray(); }
                throw new FormatException("数组缺少 , 或 ]（位置 " + i + "）");
            }
        }
        static string ParseString(string s, ref int i)
        {
            StringBuilder sb = new StringBuilder();
            i++;
            while (i < s.Length)
            {
                char c = s[i++];
                if (c == '"') return sb.ToString();
                if (c == '\\')
                {
                    if (i >= s.Length) break;
                    char e = s[i++];
                    switch (e)
                    {
                        case '"': sb.Append('"'); break;
                        case '\\': sb.Append('\\'); break;
                        case '/': sb.Append('/'); break;
                        case 'b': sb.Append('\b'); break;
                        case 'f': sb.Append('\f'); break;
                        case 'n': sb.Append('\n'); break;
                        case 'r': sb.Append('\r'); break;
                        case 't': sb.Append('\t'); break;
                        case 'u': if (i + 4 <= s.Length) { sb.Append((char)Convert.ToInt32(s.Substring(i, 4), 16)); i += 4; } break;
                        default: sb.Append(e); break;
                    }
                }
                else sb.Append(c);
            }
            throw new FormatException("字符串未闭合");
        }
        static object ParseNumber(string s, ref int i)
        {
            int st = i;
            while (i < s.Length && (char.IsDigit(s[i]) || s[i] == '-' || s[i] == '+' || s[i] == '.' || s[i] == 'e' || s[i] == 'E')) i++;
            string t = s.Substring(st, i - st);
            double d;
            if (!double.TryParse(t, NumberStyles.Float, CultureInfo.InvariantCulture, out d)) throw new FormatException("非法数字：" + t);
            return d;
        }
    }

        public static string ToJson(Frame f, List<string> warnings)
        {
            int total = f.TotalBytes();
            bool fixedOk = !f.HasVarArray();
            Segment head = null, tail = null, len = null, sum = null;
            foreach (Segment s in f.Segs)
            {
                if (s.Kind == "head") head = s;
                else if (s.Kind == "tail") tail = s;
                else if (s.Kind == "len") len = s;
                else if (s.Kind == "sum") sum = s;
            }
            StringBuilder sb = new StringBuilder();
            sb.Append("{\n");
            sb.Append("  \"name\": ").Append(Q(f.Name)).Append(",\n");
            sb.Append("  \"framing\": {\n");
            sb.Append("    \"header\": ").Append(Q(head == null ? "" : Trim(HexUtil.Normalize(head.Hex)))).Append(",\n");
            if (tail != null && tail.Bytes > 0) sb.Append("    \"tail\": ").Append(Q(Trim(HexUtil.Normalize(tail.Hex)))).Append(",\n");
            sb.Append("    \"fixed_length\": ").Append(fixedOk ? total : 0).Append(",\n");
            sb.Append("    \"minimum_length\": ").Append(1).Append(",\n");
            sb.Append("    \"maximum_length\": ").Append(fixedOk ? Math.Max(total, 1) : 65536);
            if (len != null)
            {
                sb.Append(",\n    \"length_field\": {\n");
                sb.Append("      \"offset\": ").Append(f.OffsetOf(f.Segs.IndexOf(len))).Append(",\n");
                sb.Append("      \"size\": ").Append(len.LenSize).Append(",\n");
                sb.Append("      \"endian\": ").Append(Q(len.Endian)).Append(",\n");
                sb.Append("      \"adjust\": ").Append(len.LenAdjust).Append("\n    }");
            }
            sb.Append("\n  },\n");
            sb.Append("  \"fields\": [\n");
            List<string> lines = new List<string>();
            for (int i = 0; i < f.Segs.Count; i++)
            {
                Segment s = f.Segs[i];
                if (s.Kind != "data" || !s.Export) continue;
                int w = PTypes.Width(s.PType);
                lines.Add("    {\n      \"name\": " + Q(s.Name) + ",\n      \"offset\": " + f.OffsetOf(i) + ",\n      \"type\": " + Q(s.PType)
                    + ",\n      \"endian\": " + Q(s.Endian) + ",\n      \"scale\": " + Num(s.Scale) + ",\n      \"bias\": " + Num(s.Bias) + "\n    }");
            }
            for (int i = 0; i < lines.Count; i++) sb.Append(lines[i]).Append(i == lines.Count - 1 ? "\n" : ",\n");
            sb.Append("  ],\n");
            string ct = (sum == null || !sum.Export) ? "none" : sum.ChkType;
            sb.Append("  \"checksum\": { \"type\": ").Append(Q(ct));
            if (ct != "none" && sum != null)
            {
                int off = f.OffsetOf(f.Segs.IndexOf(sum));
                int rf = sum.RangeFrom;
                int rt = sum.RangeTo < 0 ? off : sum.RangeTo;
                sb.Append(", \"offset\": ").Append(off)
                  .Append(", \"range_start\": ").Append(rf)
                  .Append(", \"range_end\": ").Append(rt)
                  .Append(", \"endian\": ").Append(Q(sum.Endian));
            }
            sb.Append(" }\n}\n");
            // 提示无法表达的部分
            foreach (Segment s in f.Segs)
            {
                if (s.Kind == "data" && s.Bytes == 3) warnings.Add("3 字节字段插件不支持，已按保留字节处理（不输出通道）");
                if (s.Kind == "resv" && (s.Tag == "定长数组" || s.Tag == "变长数组")) warnings.Add(s.Tag + " 插件不支持，仅按 " + s.Bytes + " 字节占位，保证后续字段偏移正确");
            }
            return sb.ToString();
        }

        static string Trim(string hex)
        {
            return hex == null ? "" : hex.Trim();
        }

        public static Frame FromJson(string text)
        {
            Dictionary<string, object> root = MiniJson.Parse(text) as Dictionary<string, object>;
            if (root == null) throw new FormatException("JSON 根节点不是对象");
            Frame f = new Frame();
            f.Name = S(root, "name", "MyProtocol");
            Dictionary<string, object> framing = Sub(root, "framing");
            string header = "", tail = "";
            int fixedLen = 0;
            Dictionary<string, object> lf = null;
            if (framing != null)
            {
                header = S(framing, "header", "");
                tail = S(framing, "tail", "");
                fixedLen = I(framing, "fixed_length", 0);
                lf = Sub(framing, "length_field");
            }
            Dictionary<string, object> ck = Sub(root, "checksum");
            string chkType = ck == null ? "none" : S(ck, "type", "none");
            int chkOff = ck == null ? -1 : I(ck, "offset", -1);
            int chkFrom = ck == null ? 0 : I(ck, "range_start", 0);
            int chkTo = ck == null ? -1 : I(ck, "range_end", -1);

            List<Segment> data = new List<Segment>();
            List<int> dataOff = new List<int>();
            List<string> dataName = new List<string>();
            object fo;
            if (root.TryGetValue("fields", out fo) && fo != null)
            {
                object[] arr = fo as object[];
                if (arr != null)
                {
                    foreach (object o in arr)
                    {
                        Dictionary<string, object> d = o as Dictionary<string, object>;
                        if (d == null) continue;
                        Segment s = new Segment();
                        s.Kind = "data";
                        s.Name = S(d, "name", "field");
                        s.PType = S(d, "type", "float32");
                        s.Bytes = PTypes.Width(s.PType);
                        if (s.Bytes == 0) { s.Bytes = 4; s.PType = "uint32"; }
                        s.Endian = S(d, "endian", "little");
                        s.Scale = D(d, "scale", 1.0);
                        s.Bias = D(d, "bias", 0.0);
                        data.Add(s);
                        dataOff.Add(I(d, "offset", 0));
                        dataName.Add(s.Name);
                    }
                }
            }
            // 先按偏移排序
            for (int i = 0; i < data.Count; i++)
                for (int j = i + 1; j < data.Count; j++)
                    if (dataOff[j] < dataOff[i])
                    {
                        Segment ts = data[i]; data[i] = data[j]; data[j] = ts;
                        int ti = dataOff[i]; dataOff[i] = dataOff[j]; dataOff[j] = ti;
                    }

            int headBytes = 0;
            try { headBytes = HexUtil.ByteCount(HexUtil.Normalize(header)); } catch { headBytes = 0; }
            int tailBytes = 0;
            try { tailBytes = HexUtil.ByteCount(HexUtil.Normalize(tail)); } catch { tailBytes = 0; }
            int total = fixedLen;
            if (total <= 0)
            {
                total = headBytes;
                for (int i = 0; i < data.Count; i++) if (dataOff[i] + data[i].Bytes > total) total = dataOff[i] + data[i].Bytes;
                if (chkType != "none" && chkOff >= 0 && chkOff + PTypes.ChkBytes(chkType) > total) total = chkOff + PTypes.ChkBytes(chkType);
                if (tailBytes > 0 && total < headBytes) total = headBytes + tailBytes;
                else if (tailBytes > 0 && total == headBytes) total = headBytes + tailBytes;
            }
            if (total < headBytes + tailBytes) total = headBytes + tailBytes;

            // 用"字节归属"方式重建段序列
            string[] owner = new string[total + 4];
            for (int i = 0; i < owner.Length; i++) owner[i] = "";
            if (headBytes > 0) for (int i = 0; i < headBytes && i < owner.Length; i++) owner[i] = "h";
            if (tailBytes > 0) for (int i = total - tailBytes; i < total; i++) if (i >= 0 && i < owner.Length) owner[i] = "t";
            if (lf != null)
            {
                int lo = I(lf, "offset", 0); int lsz = I(lf, "size", 2);
                for (int i = lo; i < lo + lsz && i < owner.Length; i++) if (i >= 0) owner[i] = "l";
            }
            for (int k = 0; k < data.Count; k++)
                for (int i = dataOff[k]; i < dataOff[k] + data[k].Bytes && i < owner.Length; i++)
                    if (i >= 0) owner[i] = "d" + k.ToString();
            if (chkType != "none" && chkOff >= 0)
            {
                int cb = PTypes.ChkBytes(chkType);
                for (int i = chkOff; i < chkOff + cb && i < owner.Length; i++) if (i >= 0) owner[i] = "c";
            }

            f.Segs.Clear();
            int pos = 0;
            while (pos < total)
            {
                string o = owner[pos];
                if (o == "")
                {
                    int st = pos;
                    while (pos < total && owner[pos] == "") pos++;
                    Segment r = new Segment();
                    r.Kind = "resv"; r.Tag = "保留"; r.Bytes = pos - st;
                    f.Segs.Add(r);
                    continue;
                }
                if (o == "h")
                {
                    Segment s = new Segment(); s.Kind = "head"; s.Bytes = headBytes; s.Hex = Trim(HexUtil.Normalize(header));
                    f.Segs.Add(s); pos += headBytes; continue;
                }
                if (o == "t")
                {
                    Segment s = new Segment(); s.Kind = "tail"; s.Bytes = tailBytes; s.Hex = Trim(HexUtil.Normalize(tail));
                    f.Segs.Add(s); pos += tailBytes; continue;
                }
                if (o == "l")
                {
                    Segment s = new Segment(); s.Kind = "len"; s.LenSize = I(lf, "size", 2); s.Bytes = s.LenSize;
                    s.Endian = S(lf, "endian", "little"); s.LenAdjust = I(lf, "adjust", 0);
                    f.Segs.Add(s); pos += s.Bytes; continue;
                }
                if (o == "c")
                {
                    Segment s = new Segment(); s.Kind = "sum"; s.ChkType = chkType; s.Endian = S(ck, "endian", "little");
                    s.Bytes = PTypes.ChkBytes(chkType); s.RangeFrom = chkFrom;
                    s.RangeTo = chkTo < 0 ? chkOff : chkTo;
                    f.Segs.Add(s); pos += s.Bytes; continue;
                }
                int idx = 0;
                int.TryParse(o.Substring(1), out idx);
                Segment ds = data[idx];
                ds.Endian = ds.Endian;
                f.Segs.Add(ds);
                pos += ds.Bytes;
            }
            return f;
        }

        static Dictionary<string, object> Sub(Dictionary<string, object> d, string k)
        {
            object o;
            if (d != null && d.TryGetValue(k, out o) && o != null) return o as Dictionary<string, object>;
            return null;
        }
        static string S(Dictionary<string, object> d, string k, string def)
        {
            object o;
            if (d != null && d.TryGetValue(k, out o) && o != null) return Convert.ToString(o, CultureInfo.InvariantCulture);
            return def;
        }
        static int I(Dictionary<string, object> d, string k, int def)
        {
            object o;
            if (d != null && d.TryGetValue(k, out o) && o != null) { try { return Convert.ToInt32(o, CultureInfo.InvariantCulture); } catch { return def; } }
            return def;
        }
        static double D(Dictionary<string, object> d, string k, double def)
        {
            object o;
            if (d != null && d.TryGetValue(k, out o) && o != null) { try { return Convert.ToDouble(o, CultureInfo.InvariantCulture); } catch { return def; } }
            return def;
        }
        static string Q(string s)
        {
            if (s == null) s = "";
            StringBuilder sb = new StringBuilder("\"");
            foreach (char c in s)
            {
                if (c == '"') sb.Append("\\\"");
                else if (c == '\\') sb.Append("\\\\");
                else if (c == '\n') sb.Append("\\n");
                else if (c == '\r') sb.Append("\\r");
                else if (c == '\t') sb.Append("\\t");
                else sb.Append(c);
            }
            sb.Append("\"");
            return sb.ToString();
        }
        static string Num(double v) { return v.ToString("0.############", CultureInfo.InvariantCulture); }

        public static List<string> Validate(Frame f)
        {
            List<string> e = new List<string>();
            if (f.Name == null || f.Name.Trim().Length == 0) e.Add("协议名不能为空");
            if (f.Segs.Count == 0) e.Add("至少要有一个段（帧头/字段/帧尾…）");
            int total = f.TotalBytes();
            if (total <= 0) e.Add("总帧长必须大于 0");
            Segment head = null, tail = null, len = null, sum = null;
            foreach (Segment s in f.Segs)
            {
                if (s.Kind == "head") head = s;
                if (s.Kind == "tail") tail = s;
                if (s.Kind == "len") len = s;
                if (s.Kind == "sum") sum = s;
            }
            try { HexUtil.Normalize(head == null ? "" : head.Hex); } catch (Exception ex) { e.Add("帧头非法：" + ex.Message); }
            try { HexUtil.Normalize(tail == null ? "" : tail.Hex); } catch (Exception ex) { e.Add("帧尾非法：" + ex.Message); }
            HashSet<string> names = new HashSet<string>();
            int i = 0;
            foreach (Segment s in f.Segs)
            {
                int idx = f.OffsetOf(i);
                if (s.Kind == "data")
                {
                    if (s.Name == null || s.Name.Trim().Length == 0) e.Add("第 " + (i + 1) + " 段数据字段名不能为空");
                    else if (!names.Add(s.Name)) e.Add("数据字段名重复：" + s.Name);
                    if (PTypes.Width(s.PType) != s.Bytes) e.Add("字段 " + (s.Name.Length > 0 ? s.Name : ("#" + (i + 1))) + " 类型与字节数不匹配");
                }
                if (s.Kind == "len")
                {
                    if (s.LenSize != 1 && s.LenSize != 2 && s.LenSize != 4) e.Add("帧长度字段宽度只支持 1/2/4");
                }
                if (s.Kind == "sum" && s.Export)
                {
                    if (s.ChkType != "none" && idx < 0) e.Add("校验段偏移非法");
                }
                i++;
            }
            if (f.HasVarArray() && len == null) e.Add("含变长数组时必须配置「帧长度」段，且 fixed_length 需要手工在插件里核对");
            return e;
        }

        public static List<string> Warnings(Frame f)
        {
            List<string> w = new List<string>();
            List<string> dummy = new List<string>();
            try { ToJson(f, dummy); } catch { }
            foreach (string s in dummy)
            {
                bool has = false;
                foreach (string x in w) if (x == s) has = true;
                if (!has) w.Add(s);
            }
            return w;
        }

        int dragFrom = -1;
        bool dragArmed = false;
        Point dragStart = Point.Empty;
        Timer dragTimer;

        void HookDrag()
        {
            dragTimer = new Timer();
            dragTimer.Interval = 100;
            dragTimer.Tick += delegate
            {
                dragTimer.Stop();
                if (dragFrom >= 0) { dragArmed = true; grid.Cursor = Cursors.SizeAll; }
            };
            grid.MouseDown += delegate(object o, MouseEventArgs e)
            {
                if (e.Button != MouseButtons.Left) return;
                DataGridView.HitTestInfo h = grid.HitTest(e.X, e.Y);
                try { grid.EndEdit(); } catch { } if (h.RowIndex >= 0 && h.ColumnIndex == 0)
                {
                    dragFrom = h.RowIndex;
                    dragStart = new Point(e.X, e.Y);
                    dragTimer.Start();
                }
            };
            grid.MouseMove += delegate(object o, MouseEventArgs e)
            {
                if (dragFrom < 0) return;
                if (!dragArmed)
                {
                    if (Math.Abs(e.X - dragStart.X) > 8 || Math.Abs(e.Y - dragStart.Y) > 8) { dragTimer.Stop(); dragFrom = -1; }
                    return;
                }
                DataGridView.HitTestInfo h = grid.HitTest(e.X, e.Y);
                if (h.RowIndex >= 0 && h.RowIndex != dragFrom)
                {
                    int to = h.RowIndex;
                    Segment s = frame.Segs[dragFrom];
                    frame.Segs.RemoveAt(dragFrom);
                    frame.Segs.Insert(to, s);
                    dragFrom = to;
                    RefreshRows();
                    SelectRow(to);
                }
            };
            grid.MouseUp += delegate
            {
                bool wasDrag = dragArmed;
                dragTimer.Stop();
                dragArmed = false;
                dragFrom = -1;
                grid.Cursor = Cursors.Default;
                if (wasDrag) ReloadGrid();
            };
        }

        void ApplyProps()
        {
            if (loading) return;
            int i = Sel();
            if (i < 0 || i >= frame.Segs.Count) return;
            Segment s = frame.Segs[i];
            bool needReload = false;
            if (s.Kind == "data")
            {
                s.Name = dName.Text;
                if (dType.SelectedItem != null && s.Bytes != 3) s.PType = ((TypeItem)dType.SelectedItem).Type;
                s.Endian = dEndian.SelectedIndex == 1 ? "big" : "little";
                double v;
                if (double.TryParse(dScale.Text, NumberStyles.Float, CultureInfo.InvariantCulture, out v)) s.Scale = v;
                if (double.TryParse(dBias.Text, NumberStyles.Float, CultureInfo.InvariantCulture, out v)) s.Bias = v;
                s.Export = dExport.Checked && s.Bytes != 3;
                try { grid.Rows[i].Cells[1].Value = frame.KindName(s); } catch { }
            }
            else if (s.Kind == "len")
            {
                int n = 2;
                if (lenSize.SelectedItem != null) int.TryParse(lenSize.SelectedItem.ToString(), out n);
                if (n != s.LenSize) needReload = true;
                s.LenSize = n; s.Bytes = n;
                s.Endian = lenEndian.SelectedIndex == 1 ? "big" : "little";
                s.LenAdjust = (int)lenAdjust.Value;
            }
            else if (s.Kind == "sum")
            {
                if (sumType.SelectedItem != null) s.ChkType = ((TypeItem)sumType.SelectedItem).Type;
                s.Export = sumEnable.Checked;
                if (s.ChkType == "none") s.Export = false;
                int nb = PTypes.ChkBytes(s.ChkType);
                if (nb != s.Bytes) needReload = true;
                s.Bytes = nb;
                int a;
                if (sumFrom.SelectedItem != null && int.TryParse(sumFrom.SelectedItem.ToString(), out a)) s.RangeFrom = a;
                if (sumTo.SelectedItem != null && int.TryParse(sumTo.SelectedItem.ToString(), out a)) s.RangeTo = a;
            }
            else if (s.Kind == "resv")
            {
                int n = s.Bytes;
                if (resvLen.SelectedItem != null) int.TryParse(resvLen.SelectedItem.ToString(), out n);
                if (n != s.Bytes) needReload = true;
                s.Bytes = n;
                s.ElemCount = (int)resvElems.Value;
            }
            if (needReload) ReloadGrid(); else { UpdateSummary(); RefreshRowValue(i); }
        }

        string SafeFormat(string raw, ref int caret)
        {
            int valid = 0;
            for (int i = 0; i < raw.Length && i < caret; i++) if (HexUtil.IsHexChar(raw[i])) valid++;
            StringBuilder sb = new StringBuilder();
            foreach (char c in raw) if (HexUtil.IsHexChar(c)) sb.Append(char.ToUpperInvariant(c));
            string h = sb.ToString();
            StringBuilder o = new StringBuilder();
            for (int i = 0; i < h.Length; i += 2)
            {
                if (i > 0) o.Append(' ');
                o.Append(h[i]);
                if (i + 1 < h.Length) o.Append(h[i + 1]);
            }
            string res = o.ToString();
            int pos = 0, cnt = 0;
            while (pos < res.Length && cnt < valid) { if (HexUtil.IsHexChar(res[pos])) cnt++; pos++; }
            if (pos < res.Length && res[pos] == ' ') pos++;
            caret = pos;
            return res;
        }

            catch { MessageBox.Show(this, "十六进制格式不正确：" + s.Hex, "提示"); }
            ReloadGrid();
        }

            }
        }

        public void OpenPath(string p)
        {
            try
            {
                if (!File.Exists(p)) { MessageBox.Show(this, "文件不存在：" + p, "提示"); return; }
                byte[] raw = File.ReadAllBytes(p);
                string head = Encoding.ASCII.GetString(raw, 0, Math.Min(raw.Length, 16));
                if (head.IndexOf('{') < 0)
                {
                    DialogResult r = MessageBox.Show(this,
                        "该文件无法解析为协议配置。" +
                        "现在用默认示例覆盖它吗？", "文件无法解析", MessageBoxButtons.YesNo, MessageBoxIcon.Warning);
                    if (r == DialogResult.Yes) { frame = Cfg.Default(); path = p; ReloadGrid(); SaveTo(p); }
                    return;
                }
                string text = Encoding.UTF8.GetString(raw);
                if (text.Length > 0 && text[0] == '\uFEFF') text = text.Substring(1);
                frame = Cfg.FromJson(text);
                path = p;
                ReloadGrid();
                SaveLastPath(p);
                slMsg.Text = "已载入 " + Path.GetFileName(p);
            }
            catch (Exception ex)
            {
                MessageBox.Show(this, "读取/解析失败：\r\n" + ex.Message, "错误", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        public void SaveTo(string p)
        {
            if (p == null || p.Length == 0) { DoSaveAs(); return; }
            try
            {
                string nn = AskName(frame.Name); if (nn == null) { slMsg.Text = "已取消保存"; return; } if (nn.Length > 0) frame.Name = nn; List<string> errs = Cfg.Validate(frame);
                if (errs.Count > 0)
                {
                    StringBuilder sb = new StringBuilder("配置不合法，未保存：\r\n\r\n");
                    foreach (string s in errs) sb.Append("  • ").Append(s).Append("\r\n");
                    MessageBox.Show(this, sb.ToString(), "校验失败", MessageBoxButtons.OK, MessageBoxIcon.Error);
                    return;
                }
                List<string> warn = Cfg.Warnings(frame);
                if (warn.Count > 0)
                {
                    StringBuilder sb = new StringBuilder("提示（可继续保存）：\r\n\r\n");
                    foreach (string s in warn) sb.Append("  • ").Append(s).Append("\r\n");
                    sb.Append("\r\n仍然保存？");
                    if (MessageBox.Show(this, sb.ToString(), "确认", MessageBoxButtons.YesNo, MessageBoxIcon.Information) != DialogResult.Yes) return;
                }
                List<string> w2 = new List<string>();
                string json = Cfg.ToJson(frame, w2);
                File.WriteAllText(p, json, new UTF8Encoding(false));
                path = p;
                SaveLastPath(p);
                Text = "VOFA+ 自定义协议配置工具   [" + p + "]";
                slMsg.Text = "已保存 " + DateTime.Now.ToString("HH:mm:ss") + "（VOFA+ 约 1 秒内热重载）";
                UpdateSummary();
            }
            catch (Exception ex)
            {
                MessageBox.Show(this, "保存失败：\r\n" + ex.Message, "错误", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }
